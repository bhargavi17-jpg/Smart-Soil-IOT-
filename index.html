ESP32 C++ Production Firmware

Flash this code directly via Arduino IDE to read the soil sensor and transmit data to the web app.

#include <WiFi.h>
#include <HTTPClient.h>
#include <ArduinoJson.h>

// --- Wi-Fi Credentials ---
const char* ssid = "YOUR_WIFI_SSID";
const char* password = "YOUR_WIFI_PASSWORD";

// --- Backend URL Endpoint ---
const char* serverEndpoint = "https://your-server-url.com/api/iot/update";

// --- Pin Allocations ---
const int SOIL_PIN = 34;    // Capacitive Sensor connected to ADC1 GPIO 34
const int RELAY_PIN = 26;   // 5V Relay Signal connected to GPIO 26

// Sensor Calibration Values (Calibrate in pure air vs submerged in water)
const int AIR_VALUE = 3200;    // Dry air ADC value
const int WATER_VALUE = 1350;  // Submerged in water ADC value

void setup() {
  Serial.begin(115200);
  pinMode(SOIL_PIN, INPUT);
  pinMode(RELAY_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, HIGH); // Default OFF (assuming active-LOW relay)

  WiFi.begin(ssid, password);
  Serial.print("Connecting to WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi Connected! Local IP: " + WiFi.localIP().toString());
}

void loop() {
  if (WiFi.status() == WL_CONNECTED) {
    // 1. Read Analog Soil Moisture with multi-sampling to filter noise
    long sensorSum = 0;
    for (int i = 0; i < 10; i++) {
      sensorSum += analogRead(SOIL_PIN);
      delay(20);
    }
    int rawValue = sensorSum / 10;

    // 2. Map raw analog readings to percentage
    int moisturePercent = map(rawValue, AIR_VALUE, WATER_VALUE, 0, 100);
    moisturePercent = constrain(moisturePercent, 0, 100);

    Serial.printf("Raw ADC: %d | Moisture: %d%%\n", rawValue, moisturePercent);

    // 3. Post telemetry to Node.js backend
    HTTPClient http;
    http.begin(serverEndpoint);
    http.addHeader("Content-Type", "application/json");

    StaticJsonDocument<200> doc;
    doc["moisture"] = moisturePercent;
    doc["temp"] = 31.5;      // Can be replaced with DHT22 read value
    doc["humidity"] = 65.0;  // Can be replaced with DHT22 read value

    String payload;
    serializeJson(doc, payload);

    int httpCode = http.POST(payload);
    if (httpCode > 0) {
      String response = http.getString();
      StaticJsonDocument<200> resDoc;
      deserializeJson(resDoc, response);

      // Actuate Relay based on server autopilot response
      bool pumpCommand = resDoc["pumpStatus"];
      digitalWrite(RELAY_PIN, pumpCommand ? LOW : HIGH); // LOW activates relay
    }
    http.end();
  }

  delay(5000); // Send telemetry every 5 seconds
}
